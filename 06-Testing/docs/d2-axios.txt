    // Spec TEMPORAL (D2, paso 6): se borra después de guardar docs/d2-axios.txt.
    import axios from 'axios';
    import { http, HttpResponse } from 'msw';
    import type { Coffee } from './model';
    import { url } from './coffee.api';
    import { server } from './mocks/server';
    import { coffees } from './test-data';
    
    // El mismo getCoffees que coffee.api.ts, pero "mañana": con axios en vez de fetch.
    // validateStatus: () => true para que axios no lance solo y el switch sea el mismo.
    const getCoffeesAxios = async (): Promise<Coffee[]> => {
      const response = await axios.get<{ data: Coffee[] }>(url, {
        headers: { Accept: 'application/json' },
        validateStatus: () => true,
      });
      if (response.status < 200 || response.status >= 300) {
        switch (response.status) {
          case 404:
            throw new Error('Catálogo no encontrado');
          case 503:
            throw new Error('Servicio no disponible');
          default:
            throw new Error('Error inesperado');
        }
      }
      return response.data.data;
    };
    
    const cases = [
      { status: 404, message: 'Catálogo no encontrado' },
      { status: 503, message: 'Servicio no disponible' },
      { status: 500, message: 'Error inesperado' },
    ];
    
    describe('con el spy de globalThis.fetch (O3) y axios', () => {
      it('debería devolver los cafés del campo data cuando la API responde 200', async () => {
        vi.spyOn(globalThis, 'fetch').mockResolvedValue(
          new Response(JSON.stringify({ data: coffees }), { status: 200 }),
        );
        await expect(getCoffeesAxios()).resolves.toEqual(coffees);
      });
    
      it('debería llamar a fetch una vez con "/api/coffees" y la cabecera Accept: application/json', async () => {
        const fetchSpy = vi
          .spyOn(globalThis, 'fetch')
          .mockResolvedValue(new Response(JSON.stringify({ data: coffees }), { status: 200 }));
        await getCoffeesAxios().catch(() => undefined);
        expect(fetchSpy).toHaveBeenCalledTimes(1);
      });
    
      it.each(cases)('debería rechazar con "$message" cuando la API responde $status', async ({ status, message }) => {
        vi.spyOn(globalThis, 'fetch').mockResolvedValue(new Response('{}', { status }));
        await expect(getCoffeesAxios()).rejects.toThrow(message);
      });
    
      it('debería propagar el TypeError "Failed to fetch" tal cual cuando fetch rechaza', async () => {
        vi.spyOn(globalThis, 'fetch').mockRejectedValue(new TypeError('Failed to fetch'));
        await expect(getCoffeesAxios()).rejects.toThrow('Failed to fetch');
      });
    });
    
    describe('con MSW y axios', () => {
      beforeAll(() => server.listen({ onUnhandledFrame: 'error' }));
      afterEach(() => server.resetHandlers());
      afterAll(() => server.close());
    
      it('debería devolver los cafés del campo data cuando la API responde 200', async () => {
        await expect(getCoffeesAxios()).resolves.toEqual(coffees);
      });
    
      it('debería mandar la cabecera Accept: application/json cuando pide GET /api/coffees', async () => {
        let accept: string | null = null;
        server.use(
          http.get(url, ({ request }) => {
            accept = request.headers.get('accept');
            return HttpResponse.json({ data: coffees });
          }),
        );
        await getCoffeesAxios();
        expect(accept).toBe('application/json');
      });
    
      it.each(cases)('debería rechazar con "$message" cuando la API responde $status', async ({ status, message }) => {
        server.use(http.get(url, () => HttpResponse.json({ data: coffees }, { status })));
        await expect(getCoffeesAxios()).rejects.toThrow(message);
      });
    
      it('debería propagar el TypeError "fetch failed" de Node cuando la red falla', async () => {
        server.use(http.get(url, () => HttpResponse.error()));
        await expect(getCoffeesAxios()).rejects.toThrow(TypeError);
      });
    
      it('(informativo) qué rechaza axios cuando la red falla', async () => {
        server.use(http.get(url, () => HttpResponse.error()));
        const error = await getCoffeesAxios().catch((e: unknown) => e);
        console.log('axios rechaza con:', error instanceof Error ? `${error.name}: ${error.message}` : error);
        console.log('adaptador usado por axios:', axios.defaults.adapter);
      });
    });
