import { renderHook } from '@testing-library/react';
import { useCart } from './use-cart';
import { coffees } from './test-data';

const [, , brazil] = coffees;

describe('useCart sin act (temporal)', () => {
  it('caso 2 sin act: dos add seguidos, units tendría que ser 2', () => {
    // Arrange
    const { result } = renderHook(() => useCart(coffees));

    // Act
    result.current.add(brazil);
    result.current.add(brazil);

    // Assert
    expect(result.current.units).toBe(2);
  });

  it('un solo add sin act, se lee units enseguida, tendría que ser 1', () => {
    // Arrange
    const { result } = renderHook(() => useCart(coffees));

    // Act
    result.current.add(brazil);

    // Assert
    expect(result.current.units).toBe(1);
  });
});
